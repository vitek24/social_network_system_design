@startuml
!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

Person(user, "User")

Container(reverseProxy, "Reverse proxy", "Nginx")
Container(apiGateway, "API Gateway", "Golang")

System_Boundary(commentSystem, "Comment system") {
    Container(commentService, "Comment Service", "Golang")
    ContainerDb(commentStorage, "Comment Storage", "Postgres")
}

System_Boundary(postSystem, "Post system") {
    Container(postService, "Post Service", "Golang")
    ContainerDb(postStorage, "Post Storage", "Postgres")
}


Rel(user, reverseProxy, "Create/Update/Delete comment", "REST")
Rel(reverseProxy, apiGateway, "Create/Update/Delete comment", "REST")
Rel(apiGateway, commentService, "Create/Update/Delete comment", "REST")
Rel(commentService, commentStorage, "CRUD comment", "REST")
Rel(postService, commentService, "Get comments(batch)", "gRPC")

@enduml
