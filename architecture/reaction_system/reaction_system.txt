@startuml
!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

Person(user, "User")

Container(reverseProxy, "Reverse proxy", "Nginx")
Container(apiGateway, "API Gateway", "Golang")

System_Boundary(reactionSystem, "Reaction system") {
    Container(reactionService, "Reaction Service", "Golang")
    ContainerDb(reactionStorage, "Reaction Storage", "Postgres")
}

System_Boundary(postSystem, "Post system") {
    Container(postService, "Post Service", "Golang")
    ContainerDb(postStorage, "Post Storage", "Postgres")
}


Rel(user, reverseProxy, "Create/Update/Delete reaction", "REST")
Rel(reverseProxy, apiGateway, "Create/Update/Delete reaction", "REST")
Rel(apiGateway, reactionService, "Create/Update/Delete reaction", "REST")
Rel(reactionService, reactionStorage, "CRUD reaction", "REST")
Rel(postService, reactionService, "Get reactions(batch)", "gRPC")

@enduml
