@startuml
!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Container.puml

Person(user, "User")

Container(reverseProxy, "Reverse proxy", "Nginx")
Container(apiGateway, "API Gateway", "Golang")
Container(cdn, "CDN", "Cloudflare")

System_Boundary(contentSystem, "Content system") {
    Container(contentService, "Content Service", "Golang")
    ContainerDb(contentStorage, "Content storage", "Ceph")
}

Rel(user, reverseProxy, "Get/Upload content", "REST")
Rel(reverseProxy, apiGateway, "Get/Upload content", "REST")
Rel(apiGateway, contentService, "Get/Upload content", "REST")
Rel(contentService, contentStorage, "CRUD content", "REST")
Rel(user, cdn, "Get content", "HTTPS")
Rel(cdn, contentStorage, "Pull content", "HTTPS")

@enduml
